Supabase 連線設定

Supabase SQL（無需 Auth，使用 anon key 直接讀寫）：

create table if not exists lifecycle_plan (
  id bigint generated always as identity primary key,
  created_at timestamptz default now(),
  payload jsonb not null
);
-- 允許 anon 讀寫（單人使用，無 RLS）
alter table lifecycle_plan enable row level security;
create policy "allow_all" on lifecycle_plan for all using (true) with check (true);
